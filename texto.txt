def consultar_dia(self):
        fecha = self.filtro_fecha.date().toString(
            "yyyy-MM-dd"
        )
        conexion = conectar()
        cursor = conexion.cursor()
        cursor.execute("""
            SELECT
                COUNT(*) AS total
            FROM asistencias
            WHERE fecha = ?
            AND hora_entrada IS NOT NULL
        """, (fecha,))
        entradas = cursor.fetchone()["total"]
        cursor.execute("""
            SELECT
                COUNT(*) AS total
            FROM asistencias
            WHERE fecha = ?
            AND hora_salida IS NOT NULL
        """, (fecha,))
        salidas = cursor.fetchone()["total"]
        cursor.execute("""
            SELECT
                COUNT(*) AS total
            FROM asistencias
            WHERE fecha = ?
            AND permiso_especial IS NOT NULL
            AND permiso_especial != ''
        """, (fecha,))
        permisos = cursor.fetchone()["total"]
        cursor.execute("""
            SELECT
                COUNT(*) AS total
            FROM asistencias
            WHERE fecha = ?
            AND hora_salida IS NULL
        """, (fecha,))
        dentro = cursor.fetchone()["total"]
        conexion.close()
        QMessageBox.information(
            self,
            "Asistencia del día",
            f"Fecha: {fecha}\n\n"
            f"Entradas: {entradas}\n"
            f"Salidas: {salidas}\n"
            f"Personas dentro: {dentro}\n"
            f"Permisos especiales: {permisos}"
        )
